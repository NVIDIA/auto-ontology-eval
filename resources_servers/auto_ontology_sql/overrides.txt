# Dependency overrides for this project. Applied via `[tool.uv]
# override-dependencies` for uv sync/add, and via UV_OVERRIDE for the `uv pip
# install` that NeMo Gym generates when it builds a resources server's venv.
#
# nemo-retriever declares `prometheus-fastapi-instrumentator<8,>=7.0`, and every
# release in that range pins `starlette<1.0.0`. nemo-gym needs ray[serve], which
# needs `starlette>=1.0.1`, so the two cannot be installed together as declared.
#
# The cap is stale: nemo_retriever never imports prometheus_fastapi_instrumentator
# anywhere. It imports prometheus_client, a different package, and only inside
# nemo_retriever/service/ (its HTTP service, which we do not run). Importing
# auto_ontology.retrieval.text_to_sql.main pulls in 44 nemo_retriever modules and loads
# neither the instrumentator nor starlette nor fastapi.
#
# Version 8.x resolves cleanly against starlette>=1.0.1, so lifting the cap lets
# nemo-gym and Auto Ontology share one environment. Revisit if nemo_retriever ever starts
# importing the instrumentator: a 7.x->8.x API break would then surface at
# runtime rather than during resolution.
prometheus-fastapi-instrumentator>=8

# nemo-gym 0.6.0 pins openai==2.44.0 and langchain-openai>=1.6.0 requires
# openai>=2.45.0. The pin has to win here: openai 2.45 made
# InputTokensDetails.cache_write_tokens a required field, and the inference-api
# gateway does not return it, so a 2.45+ server rejects every verify request's
# response payload. langchain-openai only ever reads that field with dict .get,
# which yields None on 2.44 rather than failing.
openai==2.44.0

# nemo-retriever is declared as an unpinned Git dependency, so the parent
# project resolves it through uv.lock while this venv -- built with `uv pip
# install -e ../..`, which ignores the lock -- resolves it to whatever HEAD is
# at build time. The two then disagree, silently: a HEAD newer than b0a4ce1
# requires embedding models to be revision-pinned and raises on the name Auto
# Ontology passes, which get_relevant_tables catches and turns into an empty
# table list. The agent keeps running and just never sees a relevant table, so
# the arm scores several points low with nothing in the results to say why.
#
# Pinned to the commit in uv.lock so this arm embeds exactly as the legacy
# harness does. Bump both together.
nemo-retriever[tabular] @ git+https://github.com/NVIDIA/NeMo-Retriever.git@b0a4ce1f295d1457c89ae01bd806966ee05c8307#subdirectory=nemo_retriever

# Mirrors the CVE overrides in the root pyproject's [tool.uv]. Repeated rather
# than inherited because this venv is built under UV_NO_CONFIG=1, which is what
# keeps the parent's openai override from clobbering the pin above -- but it
# switches these off as well. Without them the nemo-retriever commit pinned
# above drags its own exact pillow==12.2.0 / nltk==3.9.4 back in, and ray drops
# below the version the parent runs.
nltk>=3.10.3
pillow>=12.3.0
ray>=2.58.0
urllib3>=2.8.0,<3
